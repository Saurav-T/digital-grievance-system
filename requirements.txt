Django==5.1.4

# PDF & Document Generation
reportlab==4.4.1
# Used directly in server/views.py (grievance_pdf) for the admin grievance
# report, AND as the rendering engine underneath xhtml2pdf (see below).

python-docx==1.1.2
# Used by docxtpl under the hood, and directly in client/generators.py for
# sizing the logo (docx.shared.Mm).

docxtpl==0.16.2
# Renders the fixed .docx templates (media/docx_templates/*.docx) for
# notice/job downloads.

xhtml2pdf==0.2.16
# Renders notice/job PDFs directly from HTML/CSS templates
# (client/templates/client/pdf/*.html) — replaces WeasyPrint.
# Pure-Python, built on ReportLab, so it needs NO system libraries
# (no Cairo/Pango/GDK-Pixbuf) — pip install is all that's required on
# every platform, including slim Docker images.
# Note: it only supports a subset of CSS2.1 (no flexbox/grid, limited
# <hr> support, etc.) — the pdf templates were written with that in mind.

html5lib==1.1
# Explicit dependency of xhtml2pdf's HTML parser; pinning it avoids
# resolver surprises on fresh installs.

# Image handling (required for ImageField, and used by reportlab/xhtml2pdf
# for embedding raster images into generated PDFs)
Pillow==11.3.0

django-browser-reload==1.18.0
